// SPO6 scheduler demo
// Workload:
// arrivals = [0, 3, 6, 9, 12, 15]
// bursts   = [4, 4, 7, 7, 4, 4]
// Algorithms:
// 1 = FCFS
// 2 = SPN

def putc(ch of int)
    print(ch);
end

def nl()
    putc(10);
end

def emit_space()
    putc(32);
end

def print_number(n of int)
    if n == 0 then
        putc(48);
    else
        begin
            cnt = 0;
            tmp = n;
            while tmp > 0
                digits[cnt] = tmp % 10;
                cnt = cnt + 1;
                tmp = tmp / 10;
            end
            while cnt > 0
                cnt = cnt - 1;
                putc(digits[cnt] + 48);
            end
        end
end

def print_label_fcfs()
    putc(70);
    putc(67);
    putc(70);
    putc(83);
    nl();
end

def print_label_spn()
    putc(83);
    putc(80);
    putc(78);
    nl();
end

def print_pair_label(a of int, b of int)
    putc(a);
    if b != 0 then
        putc(b);
    putc(58);
    emit_space();
end

def print_vector(values of int array[64], count of int)
    i = 0;
    while i < count
        if i > 0 then
            emit_space();
        print_number(values[i]);
        i = i + 1;
    end
    nl();
end

def load_workload(arrival of int array[8], burst of int array[8], count of int array[1])
    count[0] = 6;

    arrival[0] = 0;
    arrival[1] = 3;
    arrival[2] = 6;
    arrival[3] = 9;
    arrival[4] = 12;
    arrival[5] = 15;

    burst[0] = 4;
    burst[1] = 4;
    burst[2] = 7;
    burst[3] = 7;
    burst[4] = 4;
    burst[5] = 4;
end

def enqueue_fcfs(queue of int array[8], tail_box of int array[1], task of int)
    queue[tail_box[0]] = task;
    tail_box[0] = tail_box[0] + 1;
end

def pick_fcfs(queue of int array[8], head_box of int array[1], tail_box of int array[1]) of int
    if head_box[0] < tail_box[0] then
        begin
            result = queue[head_box[0]];
            head_box[0] = head_box[0] + 1;
        end
    else
        result = 0 - 1;
end

def select_spn(count of int, state of int array[8], burst of int array[8], arrival of int array[8]) of int
    best = 0 - 1;
    i = 0;
    while i < count
        if state[i] == 1 then
            if best == 0 - 1 then
                best = i;
            else if burst[i] < burst[best] then
                best = i;
            else if burst[i] == burst[best] && arrival[i] < arrival[best] then
                best = i;
        i = i + 1;
    end
    result = best;
end

def restore_pc(task of int, saved_pc of int array[8]) of int
    result = saved_pc[task];
end

def restore_acc(task of int, saved_acc of int array[8]) of int
    result = saved_acc[task];
end

def save_context(task of int, cpu_pc of int, cpu_acc of int, saved_pc of int array[8], saved_acc of int array[8])
    saved_pc[task] = cpu_pc;
    saved_acc[task] = cpu_acc;
end

def mark_dispatch(task of int, time of int, start_time of int array[8], state of int array[8], dispatches_box of int array[1])
    if start_time[task] == 0 - 1 then
        start_time[task] = time;
    state[task] = 2;
    dispatches_box[0] = dispatches_box[0] + 1;
end

def execute_tick(task of int, time of int, timeline of int array[64], cpu_pc_box of int array[1], cpu_acc_box of int array[1], remaining of int array[8], saved_pc of int array[8], saved_acc of int array[8])
    timeline[time] = task + 1;
    cpu_acc_box[0] = cpu_acc_box[0] + (task + 1) * 100 + cpu_pc_box[0];
    cpu_pc_box[0] = cpu_pc_box[0] + 1;
    remaining[task] = remaining[task] - 1;
    save_context(task, cpu_pc_box[0], cpu_acc_box[0], saved_pc, saved_acc);
end

def finish_task(task of int, time of int, cpu_acc of int, finish_time of int array[8], checksum of int array[8], state of int array[8], completed_box of int array[1])
    finish_time[task] = time + 1;
    checksum[task] = cpu_acc;
    state[task] = 3;
    completed_box[0] = completed_box[0] + 1;
end

def run_simulation(algo of int)
    load_workload(arrival, burst, count_box);
    count = count_box[0];

    i = 0;
    while i < count
        remaining[i] = burst[i];
        start_time[i] = 0 - 1;
        finish_time[i] = 0 - 1;
        state[i] = 0;
        saved_pc[i] = 0;
        saved_acc[i] = 0;
        checksum[i] = 0;
        wait_time[i] = 0;
        turn_time[i] = 0;
        queue[i] = 0;
        i = i + 1;
    end

    time = 0;
    next_arrival = 0;
    current = 0 - 1;
    cpu_pc = 0;
    cpu_acc = 0;
    completed_box[0] = 0;
    q_head_box[0] = 0;
    q_tail_box[0] = 0;
    total_wait = 0;
    total_turn = 0;
    scheduler_calls = 0;
    dispatches_box[0] = 0;

    if algo == 1 then
        print_label_fcfs();
    else
        print_label_spn();

    print_pair_label(65, 0);
    print_vector(arrival, count);

    print_pair_label(66, 0);
    print_vector(burst, count);

    while completed_box[0] < count
        while next_arrival < count && arrival[next_arrival] == time
            state[next_arrival] = 1;
            if algo == 1 then
                enqueue_fcfs(queue, q_tail_box, next_arrival);
            next_arrival = next_arrival + 1;
        end

        scheduler_calls = scheduler_calls + 1;

        if current == 0 - 1 then
            begin
                if algo == 1 then
                    candidate = pick_fcfs(queue, q_head_box, q_tail_box);
                else
                    candidate = select_spn(count, state, burst, arrival);

                if candidate != 0 - 1 then
                    begin
                        current = candidate;
                        cpu_pc = restore_pc(current, saved_pc);
                        cpu_acc = restore_acc(current, saved_acc);
                        mark_dispatch(current, time, start_time, state, dispatches_box);
                    end
            end

        if current == 0 - 1 then
            timeline[time] = 0;
        else
            begin
                cpu_pc_box[0] = cpu_pc;
                cpu_acc_box[0] = cpu_acc;
                execute_tick(current, time, timeline, cpu_pc_box, cpu_acc_box, remaining, saved_pc, saved_acc);
                cpu_pc = cpu_pc_box[0];
                cpu_acc = cpu_acc_box[0];

                if remaining[current] == 0 then
                    begin
                        finish_task(current, time, cpu_acc, finish_time, checksum, state, completed_box);
                        current = 0 - 1;
                    end
            end
        time = time + 1;
    end

    i = 0;
    while i < count
        wait_time[i] = finish_time[i] - arrival[i] - burst[i];
        turn_time[i] = finish_time[i] - arrival[i];
        total_wait = total_wait + wait_time[i];
        total_turn = total_turn + turn_time[i];
        i = i + 1;
    end

    print_pair_label(84, 0);
    print_vector(timeline, time);

    print_pair_label(87, 0);
    print_vector(wait_time, count);

    print_pair_label(82, 0);
    print_vector(turn_time, count);

    print_pair_label(65, 87);
    print_number(total_wait / count);
    nl();

    print_pair_label(65, 84);
    print_number(total_turn / count);
    nl();

    print_pair_label(67, 0);
    print_vector(checksum, count);

    print_pair_label(73, 0);
    print_number(scheduler_calls);
    nl();

    print_pair_label(68, 0);
    print_number(dispatches_box[0]);
    nl();

    nl();
end

def main()
    run_simulation(1);
    run_simulation(2);
end
