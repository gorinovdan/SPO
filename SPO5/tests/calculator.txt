// Calculator demo (SPO3 source)
// Input example:
// 234
// %
// 10
// Output: 4

def read_number() of int
    acc = 0;
    ch = read();
    while ch == 10 || ch == 13
        ch = read();
    end
    while ch != 10 && ch != 13
        acc = acc * 10 + (ch - 48);
        ch = read();
    end
    result = acc;
end

def read_operator() of int
    ch = read();
    while ch == 10 || ch == 13
        ch = read();
    end
    result = ch;
end

    a = read_number();
    op = read_operator();
    b = read_number();

    if op == 43 then
        res = a + b;
    else if op == 45 then
        res = a - b;
    else if op == 42 then

    else if op == 47 then

    else if op == 37 then
        res = a % b;

        res = 0;
